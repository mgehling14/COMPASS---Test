Description: Returns child and adult safety plans for dv clients.
				I know I can combine this query with the assessments but
				I chose to keep them separate for future use.
*/

CREATE VIEW [dv].[vw_SAGE_05_safetyplans] AS

WITH
cte_AdultSafetyPlans AS (
SELECT cp.document_id AS Client_ProfileID
,cp.first_name + ' ' + cp.last_name AS client_name
,CASE
	WHEN pe.Program = 'CAAFA Outreach' THEN 'CAAFA Outreach'
	WHEN pe.Program = 'REACH Center' THEN 'REACH Center' --included even though they don't do assessments or safety plans
	WHEN pe.Program = 'Autumn House' THEN 'Autumn House'
	WHEN pe.Program =  'Faith House' THEN 'Faith House'
	WHEN pe.Program = 'CAAFA Shelter' THEN 'CAAFA Shelter'
	WHEN pe.Program = 'DVSTOP' THEN 'DVSTOP'
END AS [Program_Name]
,pe.ClientType
,CAST(asp.CompletionDate AS DATE) AS SafetyPlanDate
,asp.RecordID AS safetyplan_RecordID

FROM Client_Profile AS cp
	JOIN documents docs ON docs.ParentId=cp.document_id
	JOIN ProgramEnrollment pe ON pe.RecordID=docs.RecordId
	JOIN Links link ON link.LinkToId=docs.RecordId
	JOIN AdultSafetyPlan asp ON asp.RecordID=LinkFromId

WHERE cp.last_name != 'Test'
	AND pe.ProjectType NOT IN ('Transitional Housing') --FH & AH Transitional not included
	AND pe.ProgramType NOT IN ('Hotline') --safedvs not included, also checking for data entry problems where dvstop accidentally leaves 'dvstop' as program on the hotline calls
	AND pe.Funder IN ('VOCA', 'OVW-Late Life') --updated on 3.4.26 MS
),

cte_ChildSafetyPlans AS (
SELECT cp.document_id AS Client_ProfileID
,cp.first_name + ' ' + cp.last_name AS client_name
,CASE
	WHEN pe.Program = 'CAAFA Outreach' THEN 'CAAFA Outreach'
	WHEN pe.Program = 'REACH Center' THEN 'REACH Center' --included even though they don't do assessments or safety plans
	WHEN pe.Program = 'Autumn House' THEN 'Autumn House'
	WHEN pe.Program =  'Faith House' THEN 'Faith House'
	WHEN pe.Program = 'CAAFA Shelter' THEN 'CAAFA Shelter'
	WHEN pe.Program = 'DVSTOP' THEN 'DVSTOP'
END AS [Program_Name]
,pe.ClientType
,CAST(csp.PlanCompletionDate AS DATE) AS SafetyPlanDate
,csp.RecordID AS safetyplan_RecordID

FROM Client_Profile AS cp
	JOIN documents docs ON docs.ParentId=cp.document_id
	JOIN ProgramEnrollment pe ON pe.RecordID=docs.RecordId
	JOIN Links link ON link.LinkToId=docs.RecordId
	JOIN ChildSafetyPlan csp ON csp.RecordID=LinkFromId

WHERE cp.last_name != 'Test'
	AND pe.ProjectType NOT IN ('Transitional Housing') --FH & AH Transitional not included
	AND pe.ProgramType NOT IN ('Hotline') --safedvs not included, also checking for data entry problems where dvstop accidentally leaves 'dvstop' as program on the hotline calls
	AND pe.Funder IN ('VOCA', 'OVW-Late Life') --updated on 3.4.26 MS
),

cte_agg_safetyplans AS (
SELECT *
FROM cte_AdultSafetyPlans
UNION ALL
SELECT *
FROM cte_ChildSafetyPlans
)

SELECT DISTINCT *
FROM cte_agg_safetyplans agg
WHERE agg.[Program_Name] IS NOT NULL

--ORDER BY agg.Client_ProfileID, SafetyPlanDate
GO
