CREATE VIEW [dv].[vw_OVW_05_calls] AS

SELECT cp.document_id AS Client_ProfileID
,cp.first_name + ' ' + cp.last_name AS client_name
,CAST(cp.dob AS DATE) AS DOB
,cp.race
,cp.gender
,cp.ethnicity
,cp.multi_race_ind AS MultiRaceInd
,cp.mil_veteran AS VeteranStatus
,CASE
	WHEN h.[Program] IS NOT NULL THEN h.[Program] + ' DVHelpline Call'
END AS [Program_Name]
,CAST(h.DateOfContact AS DATE) AS Call_Date
,'Unknown' AS ClientType
,'Unknown' AS AbuseStatus
,'Unknown' AS VictimType
,'Unknown' AS ClientSpecialClassification
,CompleteOVWLateLifeAbuseScreening
,CallerOver50YrsOld
,AbuserIsSpousePartnerOrCloseRelative
,HasCallerExperiencedRecentDVorSVorElderAbuse

FROM Client_Profile AS cp
JOIN documents docs ON docs.ParentId=cp.document_id
JOIN Helpline h ON h.RecordID=docs.RecordId

WHERE cp.last_name NOT LIKE 'Test'
	AND h.CompleteOVWLateLifeAbuseScreening IS NOT NULL
	AND h.CompleteOVWLateLifeAbuseScreening != ''
	AND h.CompleteOVWLateLifeAbuseScreening != 'No'
GO
