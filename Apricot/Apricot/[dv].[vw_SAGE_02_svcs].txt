Description: Returns service details for clients "requesting" services
			 which really just means they were served under voca. There was no need to
			 join on the client query since we are getting services directly under VOCA.

*/

CREATE VIEW [dv].[vw_SAGE_02_svcs] AS

WITH
cte_ss AS (
SELECT cp.document_id AS Client_ProfileID
,cp.first_name + ' ' + cp.last_name AS client_name
,CASE
	WHEN svcs.ProgramEnrollment LIKE 'CAAFA Outreach%' THEN 'CAAFA Outreach'
	WHEN svcs.ProgramEnrollment LIKE 'REACH Center%' THEN 'REACH Center'
	WHEN svcs.ProgramEnrollment LIKE 'Autumn House%' THEN 'Autumn House'
	WHEN svcs.ProgramEnrollment LIKE 'Faith House%' THEN 'Faith House'
	WHEN svcs.ProgramEnrollment LIKE 'CAAFA Shelter%' THEN 'CAAFA Shelter'
	WHEN svcs.ProgramEnrollment LIKE 'DVSTOP%' THEN 'DVSTOP'
END AS [Program_Name]
,CAST(svcs.DateofService AS DATE) AS DateOfService
,svcs.InHours AS Hrs
,CASE
	WHEN svcs.DirectServiceCategories IN ('Personal Advocacy/Accompaniment') AND svcs.ServiceCategory = 'Victim Advocacy' THEN '1-Personal Advocacy' --updated on 12.3.25 to include the non-VOCA svcs. Original was just DirSvc = IndvAdv. MS
	ELSE NULL
END AS PersonalAdv --commonly referred to as Victim Advocacy for DV and Individual or Personal Advocacy for SV. MS
,CASE -- no longer needed, rolled into PAdv on 12.3.25, keeping for historical. MS
	WHEN svcs.DirectServiceCategories IN ('Information and Referral') AND svcs.InformationandReferral LIKE 'Referral%' THEN '1-Referral'
	ELSE NULL
END AS Referrals
,CASE
	WHEN (svcs.DirectServiceCategories IN ('Emotional Support or Safety Services') AND svcs.EmotionalSupportorSafetyServices IN ('Individual counseling')) OR svcs.[Service] = 'Individual Counseling' THEN '1-Counseling' --updated on 12.3.25 to include the non-VOCA svcs. Original was just DirSvc = IndvC. MS
	ELSE NULL
END AS IndvCounseling
,CASE -- no longer needed as of 12.3.25, keeping for historical. MS
	WHEN svcs.DirectServiceCategories IN ('Emotional Support or Safety Services') AND svcs.EmotionalSupportorSafetyServices LIKE 'Support Group%' AND pe.ClientType IN ('Adult') THEN '1-Adult Support Group'
	WHEN svcs.DirectServiceCategories IN ('Emotional Support or Safety Services') AND svcs.EmotionalSupportorSafetyServices LIKE 'Support Group%' AND pe.ClientType IN ('Child') THEN '1-Child Support Group'
	ELSE NULL
END AS SuppGroups
,CASE
	WHEN ((svcs.DirectServiceCategories IN ('Emotional Support or Safety Services') AND svcs.EmotionalSupportorSafetyServices LIKE 'Crisis intervention%') OR (svcs.[Service] LIKE '%Safety Planning')) AND pe.ClientType IN ('Adult') THEN '1-Adult Safety Planning' --updated on 12.3.25 to include the non-VOCA svcs. Original was just DirSvc = Crisis. MS
	WHEN ((svcs.DirectServiceCategories IN ('Emotional Support or Safety Services') AND svcs.EmotionalSupportorSafetyServices LIKE 'Crisis intervention%') OR (svcs.[Service] LIKE '%Safety Planning')) AND pe.ClientType IN ('Child') THEN '1-Child Safety Planning'	
	ELSE NULL
END AS SafetyPlan
,CASE -- no longer needed as of 12.3.25, keeping for historical. MS
	WHEN svcs.DirectServiceCategories IN ('Personal Advocacy/Accompaniment') AND svcs.PersonalAdvocacyAccompaniment IN ('Transportation Assistance') THEN '1-Transport'
	ELSE NULL
END AS Transportation
,CASE -- no longer needed as of 12.3.25, keeping for historical. MS
	WHEN svcs.[Service] = 'Issued Voucher' AND pe.ClientType IN ('Adult') THEN '1-Adult Hotel Night'
	WHEN svcs.[Service] = 'Issued Voucher' AND pe.ClientType IN ('Child') THEN '1-Child Hotel Night'
	ELSE NULL
END AS HotelNight
,CASE -- no longer needed, rolled into CMgmt on 12.3.25, keeping for historical. MS
	WHEN svcs.ServiceCategory = 'Assessment' AND svcs.[Service] = 'Adult' THEN '1-Adult Assessment'
	WHEN svcs.ServiceCategory = 'Assessment' AND svcs.[Service] = 'Child' THEN '1-Child Assessment' 
	ELSE NULL --we are including family assessments here, voca only wants adult and child specifically.
END AS Assessment
,CASE -- no longer needed, rolled into CMgmt on 12.3.25, keeping for historical. MS
	WHEN svcs.ServiceCategory = 'Intake and Discharge' AND svcs.[Service] = 'Intake' AND pe.ClientType IN ('Adult') THEN '1-Adult Intake'
	WHEN svcs.ServiceCategory = 'Intake and Discharge' AND svcs.[Service] = 'Intake' AND pe.ClientType IN ('Child') THEN '1-Child Intake'
	ELSE NULL
END AS Intake
,CASE
	WHEN svcs.ServiceCategory IN ('Case Management', 'Intake and Discharge', 'Focus Meeting', 'Crisis Intervention', 'Assessment', 'Pet Services') 
		AND svcs.DirectServiceCategories IN ('Personal Advocacy/Accompaniment') THEN '1-Case Management' --changed on 12.3.25 per VOCA/SAGE updates to reports. Original was just SvcCat = CMgmt. MS
	ELSE NULL
END AS CaseMgmt
,CASE -- no longer needed, rolled into CMgmt on 12.3.25, keeping for historical. MS
	WHEN svcs.ServiceCategory = 'FOCUS Meeting' THEN '1-Focus Plan' --the metrics are actually asking for # meetings, not # plans...confusing
	ELSE NULL
END AS FocusPlan
,svcs.RecordID AS svcs_RecordID

FROM Client_Profile AS cp
	JOIN documents docs ON docs.ParentId=cp.document_id
	JOIN ProgramEnrollment pe ON pe.RecordID=docs.RecordId
	JOIN Links link ON link.LinkToId=docs.RecordId
	JOIN [Services] svcs ON svcs.RecordID=LinkFromId

WHERE cp.last_name != 'Test'
	AND svcs.Funder IN ('VOCA', 'OVW-Late Life') --updated on 3.4.26 MS
	AND pe.ProjectType NOT IN ('Transitional Housing') --FH & AH Transitional not included, as of 2026 AH Transitional will be added. MS
	AND pe.ProgramType NOT IN ('Hotline') --safedvs not included, also checking for data entry problems where dvstop accidentally leaves 'dvstop' as program on the hotline calls
)

SELECT DISTINCT *
FROM cte_ss ss
WHERE ss.[Program_Name] IS NOT NULL

--ORDER BY ss.Client_ProfileID, [Program_Name], DateOfService
GO
